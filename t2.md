```python
class DSU:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # path compression
        return self.parent[x]
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py: return False
        if self.rank[px] < self.rank[py]: px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]: self.rank[px] += 1
        return True
```

```python
from collections import deque, defaultdict
def topoSort(n, edges):
    graph = defaultdict(list)
    indegree = [0] * n
    for u, v in edges:
        graph[u].append(v)
        indegree[v] += 1
    q = deque([i for i in range(n) if indegree[i] == 0])
    order = []
    while q:
        node = q.popleft()
        order.append(node)
        for nei in graph[node]:
            indegree[nei] -= 1
            if indegree[nei] == 0:
                q.append(nei)
    return order if len(order) == n else []  # [] = cycle

# Dijkstra
import heapq
def dijkstra(graph, src, n):
    dist = [float('inf')] * n
    dist[src] = 0
    heap = [(0, src)]
    while heap:
        d, u = heapq.heappop(heap)
        if d > dist[u]: continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(heap, (dist[v], v))
    return dist
```

```python
# Fisher-Yates Shuffle — O(n)
import random
def shuffle(arr):
    for i in range(len(arr)-1, 0, -1):
        j = random.randint(0, i)
        arr[i], arr[j] = arr[j], arr[i]

# Reservoir Sampling — pick k from stream
import random
def reservoir(stream, k):
    res = []
    for i, item in enumerate(stream):
        if i < k:
            res.append(item)
        else:
            j = random.randint(0, i)
            if j < k:
                res[j] = item
    return res

# Weighted Random Pick
import random, bisect
class WeightedPick:
    def __init__(self, weights):
        self.prefix = []
        total = 0
        for w in weights:
            total += w
            self.prefix.append(total)
        self.total = total
    def pick(self):
        t = random.random() * self.total
        return bisect.bisect_left(self.prefix, t)
```

## A. 📐 LINEAR ALGEBRA & MATRIX OPS

These come up constantly — know them cold.

| # | Problem | What to Implement |
|---|---------|-------------------|
| 1 | Matrix Multiplication | `C[i][j] = sum(A[i][k] * B[k][j])` |
| 2 | Transpose a Matrix | `T[j][i] = A[i][j]` |
| 3 | Dot Product | `sum(a[i]*b[i])` |
| 4 | Cosine Similarity | `dot(a,b) / (norm(a) * norm(b))` |
| 5 | L1 / L2 Norm | `sum(abs)` / `sqrt(sum(sq))` |
| 6 | Softmax | `exp(x - max(x)) / sum(exp(x - max(x)))` |
| 7 | Sigmoid | `1 / (1 + exp(-x))` |
| 8 | ReLU / Leaky ReLU | `max(0, x)` / `x if x>0 else ax` |
| 9 | Batch Normalization (forward) | `(x - mean) / sqrt(var + eps)` |
| 10 | Layer Normalization | same but over feature dim |

```python
import numpy as np

def matmul(A, B):
    n, m, p = len(A), len(A[0]), len(B[0])
    C = [[0]*p for _ in range(n)]
    for i in range(n):
        for k in range(m):
            for j in range(p):
                C[i][j] += A[i][k] * B[k][j]
    return C

def cosine_similarity(a, b):
    dot = sum(x*y for x,y in zip(a,b))
    norm_a = sum(x**2 for x in a)**0.5
    norm_b = sum(x**2 for x in b)**0.5
    return dot / (norm_a * norm_b + 1e-9)

def softmax(x):
    m = max(x)  # numerical stability
    e = [2.718281828**(v - m) for v in x]
    s = sum(e)
    return [v/s for v in e]

def sigmoid(x):
    return 1 / (1 + 2.718281828**(-x))

def batch_norm(x, eps=1e-8):
    # x: list of values (one feature across batch)
    mean = sum(x) / len(x)
    var = sum((v - mean)**2 for v in x) / len(x)
    return [(v - mean) / (var + eps)**0.5 for v in x]
```

## B. 📉 LOSS FUNCTIONS

| # | Loss | Formula | Use Case |
|---|------|---------|----------|
| 1 | MSE | `mean((y - ŷ)²)` | Regression |
| 2 | MAE | `mean(|y - ŷ|)` | Robust regression |
| 3 | Binary Cross-Entropy | `-mean(y*log(p) + (1-y)*log(1-p))` | Binary clf |
| 4 | Categorical Cross-Entropy | `-sum(y * log(p))` | Multi-class |
| 5 | Hinge Loss | `mean(max(0, 1 - y*ŷ))` | SVM |
| 6 | Huber Loss | MSE if small, MAE if large | Robust regression |
| 7 | KL Divergence | `sum(p * log(p/q))` | VAE, distillation |

```python
import math

def mse(y_true, y_pred):
    return sum((a-b)**2 for a,b in zip(y_true,y_pred)) / len(y_true)

def binary_cross_entropy(y_true, y_pred, eps=1e-9):
    return -sum(
        y*math.log(p+eps) + (1-y)*math.log(1-p+eps)
        for y,p in zip(y_true, y_pred)
    ) / len(y_true)

def categorical_cross_entropy(y_true, y_pred, eps=1e-9):
    # y_true: one-hot, y_pred: softmax output
    return -sum(
        y*math.log(p+eps)
        for row_t, row_p in zip(y_true, y_pred)
        for y, p in zip(row_t, row_p)
    ) / len(y_true)

def huber_loss(y_true, y_pred, delta=1.0):
    total = 0
    for a, b in zip(y_true, y_pred):
        err = abs(a - b)
        total += 0.5*err**2 if err <= delta else delta*(err - 0.5*delta)
    return total / len(y_true)

def kl_divergence(p, q, eps=1e-9):
    return sum(pi * math.log((pi+eps)/(qi+eps)) for pi,qi in zip(p,q))
```

```python
def confusion_matrix_metrics(y_true, y_pred):
    TP = sum(1 for a,b in zip(y_true,y_pred) if a==1 and b==1)
    TN = sum(1 for a,b in zip(y_true,y_pred) if a==0 and b==0)
    FP = sum(1 for a,b in zip(y_true,y_pred) if a==0 and b==1)
    FN = sum(1 for a,b in zip(y_true,y_pred) if a==1 and b==0)
    precision = TP / (TP + FP + 1e-9)
    recall    = TP / (TP + FN + 1e-9)
    f1        = 2 * precision * recall / (precision + recall + 1e-9)
    accuracy  = (TP + TN) / len(y_true)
    return {"precision": precision, "recall": recall, "f1": f1, "accuracy": accuracy}

def auc_roc(y_true, y_scores):
    # Sort by score descending
    pairs = sorted(zip(y_scores, y_true), reverse=True)
    tp = fp = 0
    total_pos = sum(y_true)
    total_neg = len(y_true) - total_pos
    auc = 0
    prev_fp = 0
    for score, label in pairs:
        if label == 1: tp += 1
        else:
            auc += tp * 1  # trapezoid: height=tp, width=1 fp step
            fp += 1
    return auc / (total_pos * total_neg + 1e-9)

def iou(box1, box2):
    # box = [x1, y1, x2, y2]
    xi1 = max(box1[0], box2[0]); yi1 = max(box1[1], box2[1])
    xi2 = min(box1[2], box2[2]); yi2 = min(box1[3], box2[3])
    inter = max(0, xi2-xi1) * max(0, yi2-yi1)
    area1 = (box1[2]-box1[0]) * (box1[3]-box1[1])
    area2 = (box2[2]-box2[0]) * (box2[3]-box2[1])
    union = area1 + area2 - inter
    return inter / (union + 1e-9)

def ndcg(relevance, k=None):
    import math
    if k: relevance = relevance[:k]
    dcg  = sum(r / math.log2(i+2) for i,r in enumerate(relevance))
    ideal = sorted(relevance, reverse=True)
    idcg = sum(r / math.log2(i+2) for i,r in enumerate(ideal))
    return dcg / (idcg + 1e-9)
```

```python
class LinearRegression:
    def __init__(self, lr=0.01, epochs=1000):
        self.lr, self.epochs = lr, epochs
        self.w = self.b = 0

    def fit(self, X, y):
        n = len(y)
        for _ in range(self.epochs):
            y_pred = [self.w * x + self.b for x in X]
            dw = -2/n * sum((y[i]-y_pred[i])*X[i] for i in range(n))
            db = -2/n * sum(y[i]-y_pred[i] for i in range(n))
            self.w -= self.lr * dw
            self.b -= self.lr * db

    def predict(self, X):
        return [self.w * x + self.b for x in X]
```

```python
class LogisticRegression:
    def __init__(self, lr=0.01, epochs=1000):
        self.lr, self.epochs = lr, epochs

    def fit(self, X, y):
        n, d = len(X), len(X[0])
        self.w = [0.0]*d; self.b = 0.0
        for _ in range(self.epochs):
            for i in range(n):
                z = sum(self.w[j]*X[i][j] for j in range(d)) + self.b
                p = 1/(1+math.exp(-z))
                err = p - y[i]
                for j in range(d):
                    self.w[j] -= self.lr * err * X[i][j]
                self.b -= self.lr * err

        return 1/(1+math.exp(-z))
```

```python
import random

def kmeans(X, k, max_iter=100):
    # Random init
    centroids = random.sample(X, k)

    for _ in range(max_iter):
        # Assign
        clusters = [[] for _ in range(k)]
        for x in X:
            dists = [sum((a-b)**2 for a,b in zip(x,c)) for c in centroids]
            clusters[dists.index(min(dists))].append(x)

        # Update
        new_centroids = []
        for cluster in clusters:
            if cluster:
                d = len(cluster[0])
                new_centroids.append([
                    sum(p[j] for p in cluster)/len(cluster)
                    for j in range(d)
                ])
            else:
                new_centroids.append(random.choice(X))

        if new_centroids == centroids:
            break
        centroids = new_centroids

    return centroids, clusters
```

```python
import math
from collections import defaultdict

class NaiveBayes:
    def fit(self, X, y):
        self.classes = set(y)
        self.priors = {}
        self.stats = {}  # class -> feature -> (mean, std)
        n = len(y)
        for c in self.classes:
            X_c = [X[i] for i in range(n) if y[i]==c]
            self.priors[c] = len(X_c)/n
            self.stats[c] = []
            for j in range(len(X[0])):
                col = [x[j] for x in X_c]
                mean = sum(col)/len(col)
                std = (sum((v-mean)**2 for v in col)/len(col))**0.5
                self.stats[c].append((mean, std))

    def _gaussian(self, x, mean, std, eps=1e-9):
        exp = math.exp(-0.5*((x-mean)/(std+eps))**2)
        return exp / ((std+eps) * (2*math.pi)**0.5)

    def predict(self, x):
        best_class, best_prob = None, float('-inf')
        for c in self.classes:
            log_prob = math.log(self.priors[c])
            for j, (mean, std) in enumerate(self.stats[c]):
                log_prob += math.log(self._gaussian(x[j], mean, std) + 1e-9)
            if log_prob > best_prob:
                best_prob = log_prob; best_class = c
        return best_class
```

```python
class SGDMomentum:
    def __init__(self, lr=0.01, momentum=0.9):
        self.lr = lr; self.momentum = momentum; self.velocity = None

    def step(self, params, grads):
        if self.velocity is None:
            self.velocity = [0.0]*len(params)

            self.velocity[i] = self.momentum*self.velocity[i] - self.lr*grads[i]
            params[i] += self.velocity[i]
        return params

# Adam optimizer
class Adam:
    def __init__(self, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr=lr; self.beta1=beta1; self.beta2=beta2; self.eps=eps
        self.m=None; self.v=None; self.t=0

    def step(self, params, grads):
        if self.m is None:
            self.m = [0.0]*len(params)
            self.v = [0.0]*len(params)
        self.t += 1
        for i in range(len(params)):
            self.m[i] = self.beta1*self.m[i] + (1-self.beta1)*grads[i]
            self.v[i] = self.beta2*self.v[i] + (1-self.beta2)*grads[i]**2
            m_hat = self.m[i] / (1-self.beta1**self.t)
            v_hat = self.v[i] / (1-self.beta2**self.t)
            params[i] -= self.lr * m_hat / (v_hat**0.5 + self.eps)
        return params
```

```python
# 2D Convolution (no padding, stride=1)
def conv2d(image, kernel):
    ih, iw = len(image), len(image[0])
    kh, kw = len(kernel), len(kernel[0])
    oh, ow = ih - kh + 1, iw - kw + 1
    output = [[0.0]*ow for _ in range(oh)]
    for i in range(oh):
        for j in range(ow):
            for ki in range(kh):
                for kj in range(kw):
                    output[i][j] += image[i+ki][j+kj] * kernel[ki][kj]
    return output

# Max Pooling (2x2, stride=2)
def max_pool2d(x, pool_size=2, stride=2):
    ih, iw = len(x), len(x[0])
    oh = (ih - pool_size) // stride + 1
    ow = (iw - pool_size) // stride + 1
    out = [[0.0]*ow for _ in range(oh)]
    for i in range(oh):
        for j in range(ow):
            patch = [x[i*stride+ki][j*stride+kj]
                     for ki in range(pool_size) for kj in range(pool_size)]
            out[i][j] = max(patch)
    return out

# Scaled Dot-Product Self-Attention
def self_attention(Q, K, V):
    import math
    d_k = len(Q[0])
    # scores = Q @ K^T / sqrt(d_k)
    scores = []
    for q in Q:
        row = []
        for k in K:
            row.append(sum(qi*ki for qi,ki in zip(q,k)) / math.sqrt(d_k))
        scores.append(row)
    # softmax each row
    attn = [softmax(row) for row in scores]
    # output = attn @ V
    out = []
    for row in attn:
        out.append([sum(row[i]*V[i][j] for i in range(len(V)))
                    for j in range(len(V[0]))])
    return out

def positional_encoding(seq_len, d_model):

            if i+1 < d_model:
```

```python
def tfidf(docs):
    import math
    # Term frequency
    tf = []
    for doc in docs:
        freq = {}
        for w in doc: freq[w] = freq.get(w,0) + 1
        tf.append({w: c/len(doc) for w,c in freq.items()})
    # Inverse document frequency
    N = len(docs)
    vocab = set(w for doc in docs for w in doc)
    idf = {w: math.log(N / (1 + sum(1 for d in tf if w in d)))
           for w in vocab}
    # TF-IDF
    return [{w: tf_val * idf[w] for w, tf_val in doc.items()} for doc in tf]

def jaccard(set_a, set_b):
    a, b = set(set_a), set(set_b)
    return len(a & b) / len(a | b) if a | b else 0

def levenshtein(s1, s2):
    m, n = len(s1), len(s2)
    dp = [[0]*(n+1) for _ in range(m+1)]
    for i in range(m+1): dp[i][0] = i
    for j in range(n+1): dp[0][j] = j
    for i in range(1,m+1):
        for j in range(1,n+1):
            if s1[i-1]==s2[j-1]: dp[i][j] = dp[i-1][j-1]
            else: dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[m][n]
```

```python
def online_mean_variance(stream):
    """Welford's online algorithm"""
    n = 0; mean = 0.0; M2 = 0.0
    for x in stream:
        n += 1
        delta = x - mean
        mean += delta / n
        M2 += delta * (x - mean)
    variance = M2 / (n-1) if n > 1 else 0
    return mean, variance

def median_of_stream():
    import heapq
    lo = []  # max-heap (negate)
    hi = []  # min-heap
    def add(num):
        heapq.heappush(lo, -num)
        heapq.heappush(hi, -heapq.heappop(lo))
        if len(hi) > len(lo):
            heapq.heappush(lo, -heapq.heappop(hi))
    def get_median():
        if len(lo) > len(hi): return -lo[0]
        return (-lo[0] + hi[0]) / 2
    return add, get_median
```

```python
# Non-Maximum Suppression
def nms(boxes, scores, iou_threshold=0.5):
    # boxes: [[x1,y1,x2,y2], ...], scores: [float, ...]
    order = sorted(range(len(scores)), key=lambda i: scores[i], reverse=True)
    keep = []
    while order:
        i = order.pop(0)
        keep.append(i)
        order = [j for j in order if iou(boxes[i], boxes[j]) < iou_threshold]
    return keep

# Beam Search (token generation)
def beam_search(start_token, get_next_probs, beam_width=3, max_len=10):
    import math
    beams = [(0.0, [start_token])]  # (neg_log_prob, tokens)
    for _ in range(max_len):
        candidates = []
        for score, seq in beams:
            probs = get_next_probs(seq[-1])  # returns [(token, prob), ...]
            for token, prob in probs:
                candidates.append((score - math.log(prob+1e-9), seq+[token]))
        beams = sorted(candidates)[:beam_width]
    return beams[0][1]  # best sequence

def temperature_sample(logits, temperature=1.0):

    probs = softmax(scaled)
    r = random.random()
    cumsum = 0
    for i, p in enumerate(probs):
        cumsum += p
        if r <= cumsum: return i
    return len(probs)-1

# MAP@K
def map_at_k(relevant_docs, retrieved_docs, k):
    hits = 0; precision_sum = 0
    for i, doc in enumerate(retrieved_docs[:k]):
        if doc in relevant_docs:
            hits += 1
            precision_sum += hits / (i+1)
    return precision_sum / min(len(relevant_docs), k)
```

```python
def entropy(labels):
    n = len(labels)
    freq = {}
    for l in labels: freq[l] = freq.get(l,0)+1
    return -sum((c/n)*math.log2(c/n+1e-9) for c in freq.values())

def gini(labels):
    n = len(labels)
    freq = {}
    for l in labels: freq[l] = freq.get(l,0)+1
    return 1 - sum((c/n)**2 for c in freq.values())

def info_gain(parent, left, right):
    n = len(parent)
    weighted = (len(left)/n)*entropy(left) + (len(right)/n)*entropy(right)
    return entropy(parent) - weighted
```

### L. BACKPROPAGATION & AUTOGRAD

| # | What | Core Idea |
|---|------|-----------|
| 1 | Forward pass (MLP) | Linear → activation, layer by layer |
| 2 | Backward pass | Chain rule, dL/dW = dL/dout × dout/dW |
| 3 | Gradient of MSE | `2*(pred - y) / n` |
| 4 | Gradient of BCE | `(pred - y) / (pred*(1-pred))` |
| 5 | Gradient of Softmax+CE | `pred - y_onehot` (combined, very clean) |
| 6 | Gradient of ReLU | `1 if x > 0 else 0` |
| 7 | Gradient of Sigmoid | `s*(1-s)` |
| 8 | Gradient of tanh | `1 - tanh(x)²` |
| 9 | Vanishing gradient (explain + fix) | Residuals / BatchNorm / careful init |
| 10 | Implement Value class (micrograd-style) | Store val + grad, define backward |

```python
# Full MLP forward + backward (1 hidden layer, from scratch)
import math

def relu(x): return max(0, x)
def relu_grad(x): return 1 if x > 0 else 0

class MLP:
    def __init__(self, n_in, n_hidden, n_out, lr=0.01):
        import random
        def init(rows, cols):
            return [[random.gauss(0, 0.1) for _ in range(cols)] for _ in range(rows)]
        self.W1 = init(n_hidden, n_in);  self.b1 = [0.0]*n_hidden
        self.W2 = init(n_out, n_hidden); self.b2 = [0.0]*n_out
        self.lr = lr

    def forward(self, x):
        # hidden layer
        self.x = x
        self.z1 = [sum(self.W1[i][j]*x[j] for j in range(len(x)))+self.b1[i]
                   for i in range(len(self.W1))]
        self.a1 = [relu(z) for z in self.z1]
        # output layer
        self.z2 = [sum(self.W2[i][j]*self.a1[j] for j in range(len(self.a1)))+self.b2[i]
                   for i in range(len(self.W2))]
        return self.z2

    def backward(self, y_true):
        n_out = len(self.z2)
        # dL/dz2 = 2*(pred-y)/n (MSE grad)
        dz2 = [(self.z2[i]-y_true[i])*2/n_out for i in range(n_out)]
        # update W2, b2
        for i in range(len(self.W2)):
            for j in range(len(self.W2[0])):
                self.W2[i][j] -= self.lr * dz2[i] * self.a1[j]
            self.b2[i] -= self.lr * dz2[i]
        # dL/da1
        da1 = [sum(self.W2[i][j]*dz2[i] for i in range(n_out))
               for j in range(len(self.a1))]
        # dL/dz1 (through ReLU)
        dz1 = [da1[i]*relu_grad(self.z1[i]) for i in range(len(self.z1))]
        # update W1, b1
        for i in range(len(self.W1)):
            for j in range(len(self.W1[0])):
                self.W1[i][j] -= self.lr * dz1[i] * self.x[j]
            self.b1[i] -= self.lr * dz1[i]
```

### M. WEIGHT INITIALIZATION

| # | Method | Formula | Why |
|---|--------|---------|-----|
| 1 | Zero init | all 0s | ❌ symmetry problem |
| 2 | Random normal | `N(0, 0.01)` | works for shallow |
| 3 | Xavier / Glorot | `N(0, sqrt(2/(fan_in+fan_out)))` | sigmoid/tanh |
| 4 | He init | `N(0, sqrt(2/fan_in))` | ReLU networks |
| 5 | Orthogonal init | random orthogonal matrix | RNNs |

```python
import math, random

def xavier_init(rows, cols):
    std = math.sqrt(2 / (rows + cols))
    return [[random.gauss(0, std) for _ in range(cols)] for _ in range(rows)]

def he_init(rows, cols):
    std = math.sqrt(2 / cols)  # fan_in = cols (input dim)
    return [[random.gauss(0, std) for _ in range(cols)] for _ in range(rows)]
```

### N. NORMALIZATION LAYERS

| # | Layer | Normalize Over | Use When |
|---|-------|---------------|----------|
| 1 | Batch Norm | batch dimension (per feature) | CNNs, large batches |
| 2 | Layer Norm | feature dimension (per sample) | Transformers, NLP |
| 3 | Instance Norm | spatial dims (per sample per channel) | Style transfer |
| 4 | Group Norm | groups of channels | Small batch CNNs |
| 5 | RMS Norm | like LayerNorm without mean subtract | LLaMA, modern LLMs |

```python
def batch_norm(X, gamma=1.0, beta=0.0, eps=1e-8):
    # X: list of values across batch (single feature)
    mean = sum(X) / len(X)
    var  = sum((x-mean)**2 for x in X) / len(X)
    x_hat = [(x-mean)/(var+eps)**0.5 for x in X]
    return [gamma*x + beta for x in x_hat]

def layer_norm(x, gamma=None, beta=None, eps=1e-8):
    # x: single sample, all features
    mean = sum(x)/len(x)
    var  = sum((v-mean)**2 for v in x)/len(x)
    x_hat = [(v-mean)/(var+eps)**0.5 for v in x]
    if gamma and beta:
        return [g*v+b for v,g,b in zip(x_hat,gamma,beta)]
    return x_hat

def rms_norm(x, gamma=None, eps=1e-8):
    rms = (sum(v**2 for v in x)/len(x) + eps)**0.5
    x_hat = [v/rms for v in x]
    return [g*v for g,v in zip(gamma,x_hat)] if gamma else x_hat
```

| # | Component | What to implement |
|---|-----------|------------------|
| 1 | Conv2D (valid, stride=1) | Slide kernel, dot product |
| 2 | Conv2D with padding | Pad input first |
| 3 | Depthwise conv | One kernel per channel |
| 4 | MaxPool2D | Slide window, take max |

```python
def pad2d(image, pad):
    h, w = len(image), len(image[0])
    padded = [[0.0]*(w+2*pad) for _ in range(h+2*pad)]
    for i in range(h):
        for j in range(w):
            padded[i+pad][j+pad] = image[i][j]
    return padded

def conv2d(image, kernel, stride=1, padding=0):
    if padding: image = pad2d(image, padding)
    ih, iw = len(image), len(image[0])
    kh, kw = len(kernel), len(kernel[0])
    oh = (ih-kh)//stride + 1
    ow = (iw-kw)//stride + 1
    out = [[0.0]*ow for _ in range(oh)]
    for i in range(oh):
        for j in range(ow):
            for ki in range(kh):
                for kj in range(kw):
                    out[i][j] += image[i*stride+ki][j*stride+kj]*kernel[ki][kj]
    return out

def avg_pool2d(x, pool=2, stride=2):
    oh = (len(x)-pool)//stride+1
    ow = (len(x[0])-pool)//stride+1
    return [[sum(x[i*stride+ki][j*stride+kj]
                 for ki in range(pool) for kj in range(pool))/(pool*pool)
             for j in range(ow)] for i in range(oh)]

def global_avg_pool(x):
    # x: 2D feature map
    total = sum(x[i][j] for i in range(len(x)) for j in range(len(x[0])))
    return total / (len(x) * len(x[0]))
```

### P. RNN / LSTM / GRU

| # | Component | Key Formula |
|---|-----------|------------|
| 1 | Vanilla RNN | `h_t = tanh(W_h @ h_{t-1} + W_x @ x_t + b)` |
| 2 | LSTM | 4 gates: forget, input, cell, output |
| 3 | GRU | 2 gates: reset, update (lighter than LSTM) |
| 4 | Bidirectional RNN | Forward + backward, concat hidden states |
| 5 | BPTT | Backprop through time, unroll k steps |

```python
import math

def tanh(x): return (math.exp(x)-math.exp(-x))/(math.exp(x)+math.exp(-x))
def sigmoid(x): return 1/(1+math.exp(-x))

def rnn_step(x, h_prev, Wx, Wh, b):
    # x: input, h_prev: prev hidden
    z = [sum(Wx[i][j]*x[j] for j in range(len(x))) +
         sum(Wh[i][j]*h_prev[j] for j in range(len(h_prev))) + b[i]
         for i in range(len(b))]
    return [tanh(v) for v in z]

def lstm_step(x, h_prev, c_prev, params):
    Wf,Wi,Wc,Wo, bf,bi,bc,bo = params
    def gate(W, b, act):
        z = [sum(W[i][j]*(x+h_prev)[j] for j in range(len(x)+len(h_prev)))+b[i]
             for i in range(len(b))]
        return [act(v) for v in z]
    f = gate(Wf, bf, sigmoid)          # forget gate
    i = gate(Wi, bi, sigmoid)          # input gate
    c_tilde = gate(Wc, bc, tanh)       # candidate cell
    o = gate(Wo, bo, sigmoid)          # output gate
    c = [f[j]*c_prev[j] + i[j]*c_tilde[j] for j in range(len(f))]
    h = [o[j]*tanh(c[j]) for j in range(len(o))]
    return h, c

def gru_step(x, h_prev, Wr, Wz, Wh, br, bz, bh):
    xh = x + h_prev
    r = [sigmoid(sum(Wr[i][j]*xh[j] for j in range(len(xh)))+br[i])
         for i in range(len(br))]
    z = [sigmoid(sum(Wz[i][j]*xh[j] for j in range(len(xh)))+bz[i])
         for i in range(len(bz))]
    xrh = x + [r[j]*h_prev[j] for j in range(len(r))]
    h_tilde = [tanh(sum(Wh[i][j]*xrh[j] for j in range(len(xrh)))+bh[i])
               for i in range(len(bh))]
    h = [(1-z[j])*h_prev[j] + z[j]*h_tilde[j] for j in range(len(z))]
    return h
```

### Q. TRANSFORMER COMPONENTS

The most important DL section for Google MLE right now.

| # | Component | Formula / Key |
|---|-----------|--------------|
| 1 | Scaled dot-product attention | `softmax(QK^T / sqrt(d_k)) @ V` |
| 2 | Multi-head attention | Split into h heads, concat, project |
| 3 | Positional encoding | `sin/cos` at increasing wavelengths |
| 4 | Feed-forward block | `Linear → ReLU → Linear` |
| 5 | Encoder block | `Attention → Add&Norm → FFN → Add&Norm` |
| 6 | Decoder (masked attention) | Causal mask: upper triangle = -inf |
| 7 | KV Cache (concept) | Store past K,V to avoid recomputation |
| 8 | Rotary Positional Embedding (RoPE) | Rotate Q,K by position angle |
| 9 | Flash Attention (concept) | Tile Q,K,V to avoid O(n²) memory |
| 10 | Cross-attention | Q from decoder, K/V from encoder |

```python
import math

def softmax(x):
    m = max(x)
    e = [math.exp(v-m) for v in x]
    s = sum(e); return [v/s for v in e]

def matmul(A, B):
    n,m,p = len(A),len(A[0]),len(B[0])
    return [[sum(A[i][k]*B[k][j] for k in range(m)) for j in range(p)] for i in range(n)]

def transpose(A):
    return [[A[i][j] for i in range(len(A))] for j in range(len(A[0]))]

def scaled_dot_product_attention(Q, K, V, mask=None):
    d_k = len(Q[0])
    # scores = Q @ K^T / sqrt(d_k)
    scores = matmul(Q, transpose(K))
    scores = [[v/math.sqrt(d_k) for v in row] for row in scores]
    if mask:  # causal mask: set upper triangle to -inf
        for i in range(len(scores)):
            for j in range(len(scores[0])):
                if mask[i][j]: scores[i][j] = float('-inf')
    attn = [softmax(row) for row in scores]
    return matmul(attn, V)

def causal_mask(seq_len):
    # 1 = mask out (future tokens)
    return [[1 if j > i else 0 for j in range(seq_len)] for i in range(seq_len)]

def multi_head_attention(Q, K, V, n_heads):
    d_model = len(Q[0])
    d_k = d_model // n_heads
    # Split into heads (simplified: slice feature dim)
    outputs = []
    for h in range(n_heads):
        Q_h = [[q[h*d_k:(h+1)*d_k] for q in Q]]
        K_h = [[k[h*d_k:(h+1)*d_k] for k in K]]
        V_h = [[v[h*d_k:(h+1)*d_k] for v in V]]
        outputs.append(scaled_dot_product_attention(Q_h[0], K_h[0], V_h[0]))
    # Concat heads
    return [[v for head_out in [outputs[h][i] for h in range(n_heads)] for v in head_out]
            for i in range(len(Q))]

def feed_forward(x, W1, b1, W2, b2):
    # x: [d_model], W1: [d_ff x d_model], W2: [d_model x d_ff]
    h = [max(0, sum(W1[i][j]*x[j] for j in range(len(x)))+b1[i])
         for i in range(len(W1))]   # ReLU
    return [sum(W2[i][j]*h[j] for j in range(len(h)))+b2[i]
            for i in range(len(W2))]
```

### R. REGULARIZATION TECHNIQUES

| # | Technique | Implement |
|---|-----------|-----------|
| 1 | L1 regularization | Add `λ * sum(|w|)` to loss; grad += `λ * sign(w)` |
| 2 | L2 regularization | Add `λ * sum(w²)` to loss; grad += `2λ * w` |
| 3 | Dropout | Zero + scale at train time (see Section E) |
| 4 | Early stopping | Track val loss, stop when it rises N epochs |
| 5 | Label smoothing | `y_smooth = (1-ε)*y + ε/K` |
| 6 | Gradient clipping | `grad = grad * (max_norm / ||grad||)` if over |

```python
def l2_loss(params, lambda_=0.01):
    return lambda_ * sum(w**2 for layer in params for row in layer for w in row)

def l2_grad(w, lambda_=0.01):
    return 2 * lambda_ * w

def clip_gradients(grads, max_norm=1.0):
    total_norm = sum(g**2 for g in grads)**0.5
    if total_norm > max_norm:
        scale = max_norm / (total_norm + 1e-8)
        return [g * scale for g in grads]
    return grads

def label_smoothing(y_onehot, epsilon=0.1):
    K = len(y_onehot)
    return [(1-epsilon)*y + epsilon/K for y in y_onehot]
```

### T. GENERATIVE MODELS (Conceptual + Code)

| # | Model | What to implement |
|---|-------|------------------|
| 1 | VAE encoder | Mean + log_var → reparameterize → z |
| 2 | VAE loss | Recon loss + KL divergence |
| 3 | GAN generator step | Maximize log(D(G(z))) |
| 4 | GAN discriminator step | Maximize log(D(x)) + log(1-D(G(z))) |
| 5 | DDPM forward (add noise) | `x_t = sqrt(α_t)*x_0 + sqrt(1-α_t)*ε` |
| 6 | DDPM reverse (denoise step) | Predict ε, subtract scaled, add noise |

```python
import math, random

# VAE reparameterization trick
def reparameterize(mean, log_var):
    std = math.exp(0.5 * log_var)
    eps = random.gauss(0, 1)
    return mean + eps * std

# VAE loss = reconstruction + KL
def vae_loss(x, x_recon, mean, log_var):
    # Reconstruction: MSE
    recon = sum((a-b)**2 for a,b in zip(x, x_recon)) / len(x)
    # KL divergence: -0.5 * sum(1 + log_var - mean² - exp(log_var))
    kl = -0.5 * sum(1 + lv - m**2 - math.exp(lv)
                    for m, lv in zip(mean, log_var))
    return recon + kl

# DDPM forward process (add noise at timestep t)
def ddpm_forward(x0, t, alphas_cumprod):
    alpha_t = alphas_cumprod[t]
    eps = [random.gauss(0,1) for _ in x0]
    x_t = [math.sqrt(alpha_t)*x + math.sqrt(1-alpha_t)*e
            for x,e in zip(x0, eps)]
    return x_t, eps
```
